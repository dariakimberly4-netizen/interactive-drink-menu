const statuses=["Received","Preparing","Ready","Completed"];
function renderOrders(){
  const list=state.orders;
  $("#pickupQueue").innerHTML=list.length?list.map((o,idx)=>`
    <div class="order-box">
      <div style="display:flex;justify-content:space-between;gap:10px"><div><b>${o.id} — ${o.name}</b><br><small style="color:var(--muted)">${o.created} • ${o.time}</small></div><b>${peso(o.total)}</b></div>

    </div>`).join(""):'<p style="color:var(--muted)">No pickup orders yet.</p>';

window.advanceOrder=i=>{
  const o=state.orders[i],idx=statuses.indexOf(o.status);
  if(idx<statuses.length-1)o.status=statuses[idx+1];
  save();renderOrders();updateAnalytics();toast("Order status updated");
};


function formatTrialDate(ms){
  return new Date(ms).toLocaleDateString([], {year:"numeric",month:"short",day:"numeric"});
}

function getTrialStatus(){
  const start = Number(state.trialStartedAt) || Date.now();
  const end = start + TRIAL_LENGTH_MS;
  const remaining = end - Date.now();
  return {start,end,remaining,expired: remaining <= 0 && !state.subscriptionActive};
}

function updateTrialUI(){
  const s = getTrialStatus();
  const badge = $("#trialBadge");

  if (state.subscriptionActive){
    badge.textContent = "✓ Subscription Active";
    badge.classList.remove("warning");
    $("#trialExpired").classList.remove("open");
    return;
  }

  if (s.remaining <= 0){
    badge.textContent = "Trial Expired";
    badge.classList.add("warning");
    $("#trialStartedOn").textContent = formatTrialDate(s.start);
    $("#trialEndedOn").textContent = formatTrialDate(s.end);
    $("#trialExpired").classList.add("open");

    // Close app-level panels so the expiration screen is the active state.
    $("#cartDrawer").classList.remove("open");
    $("#adminPanel").classList.remove("open");
    $("#productModal").classList.remove("open");
    $("#backdrop").classList.remove("open");
    return;
  }

  const d = Math.floor(s.remaining / 86400000);
  const h = Math.floor((s.remaining % 86400000) / 3600000);
  const m = Math.floor((s.remaining % 3600000) / 60000);
  badge.textContent = `🎁 Free Trial: ${d}d ${h}h ${m}m`;
  badge.classList.toggle("warning", s.remaining <= (3 * 86400000));
  $("#trialExpired").classList.remove("open");
}

  $("#subscriptionModal").classList.add("open");
}
