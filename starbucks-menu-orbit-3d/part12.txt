$("#revealAdd").addEventListener("click",()=>{
  if(!revealProduct) return;
  const p = revealProduct;
  const item = {
    ...p,
    unitPrice:p.price,
    custom:{size:"M",temp:p.cat==="Iced Coffee"?"Iced":"Hot",milk:"Regular",extra:"None"},
    qty:1
  };
  const sig=item.id+JSON.stringify(item.custom);
  const existing=state.cart.find(x=>x.sig===sig);
  if(existing) existing.qty++;
  else state.cart.push({...item,sig});
  state.analytics.adds++;
  save();
  renderCart();
  updateAnalytics();
  toast("Added to cart");
});

let dragging=false,lastX=0;
$("#orbit").addEventListener("pointerdown",e=>{
  if(e.target.closest(".product")) return;
  dragging=true;
  lastX=e.clientX;
  $("#orbit").setPointerCapture(e.pointerId);
});
$("#orbit").addEventListener("pointermove",e=>{
  if(!dragging)return;
  angleOffset+=(e.clientX-lastX)*.01;
  lastX=e.clientX;
  renderOrbit();
});
$("#orbit").addEventListener("pointerup",()=>dragging=false);
$("#orbit").addEventListener("pointercancel",()=>dragging=false);

$("#productViewer").addEventListener("pointerdown",e=>{
  dragging3D = true;
  last3DX = e.clientX;
  last3DY = e.clientY;
  $("#productViewer").classList.add("dragging");
  $("#productViewer").setPointerCapture(e.pointerId);
});
$("#productViewer").addEventListener("pointermove",e=>{
  if(!dragging3D) return;
  const dx = e.clientX - last3DX;
  const dy = e.clientY - last3DY;
  productRotateY += dx * 0.45;
  productTiltX = Math.max(-16, Math.min(18, productTiltX - dy * 0.15));
  last3DX = e.clientX;
  last3DY = e.clientY;
  apply3DProductTransform();
});
$("#productViewer").addEventListener("pointerup",()=>{
  dragging3D = false;
  $("#productViewer").classList.remove("dragging");
});
$("#productViewer").addEventListener("pointercancel",()=>{
  dragging3D = false;
  $("#productViewer").classList.remove("dragging");
});
$("#productViewer").addEventListener("dblclick", reset3DProductPosition);
