function showOrbitReveal(p){
  if(!isAvailable(p)){toast("This item is currently sold out.");return;}
  revealProduct = p;
  clearTimeout(orbitRevealTimer);

  const panel = $("#orbitReveal");
  panel.classList.remove("revealed");
  panel.classList.add("open");

  $("#revealFlatImg").src = p.image;
  $("#revealFlatImg").alt = p.name;
  $("#revealArt").style.backgroundImage = `url("${p.image}")`;
  $("#revealPriceText").textContent = peso(p.price);
  $("#revealNameText").textContent = p.name;

  const pal = revealPalette(p);
  $("#revealLiquid").style.setProperty("--rtop",pal.top);
  $("#revealLiquid").style.setProperty("--rbottom",pal.bottom);
  $("#revealLiquidBack").style.setProperty("--rtop",pal.top);
  $("#revealLiquidBack").style.setProperty("--rbottom",pal.bottom);

  // First show the selected product as a completely flat image on the surface.
  // Then rise it into the 3D cup and reveal the price.
  orbitRevealTimer = setTimeout(()=>panel.classList.add("revealed"),650);

  state.analytics.views++;
  save();
  updateAnalytics();
}

function closeOrbitReveal(){
  clearTimeout(orbitRevealTimer);
  $("#orbitReveal").classList.remove("revealed","open");
  revealProduct = null;
}

let productRotateY = -18;
let productTiltX = 4;

function shortTag(name){
  const cleaned = name.replace(/[^A-Za-z0-9 ]/g,"").trim().toUpperCase();
  return cleaned.length > 11 ? cleaned.slice(0,11) : cleaned;
}

function productPalette(name, cat){
  const key = (name + " " + cat).toLowerCase();
  if (key.includes("espresso")) return {top:"#6f472d", bottom:"#2f1a10", foam:false, iced:false};
  if (key.includes("caramel")) return {top:"#c58a4d", bottom:"#6e3d1f", foam:true, iced:key.includes("iced")};
  if (key.includes("mocha")) return {top:"#8b5a45", bottom:"#4a2a22", foam:true, iced:false};
  if (key.includes("matcha")) return {top:"#8dbb70", bottom:"#4e6e38", foam:true, iced:false};
  if (key.includes("cold brew")) return {top:"#725238", bottom:"#23150f", foam:false, iced:true};
  if (key.includes("latte")) return {top:"#cda378", bottom:"#8b593a", foam:true, iced:key.includes("iced")};
  if (key.includes("cappuccino")) return {top:"#d2b08b", bottom:"#7a4a2d", foam:true, iced:false};
  return {top:"#b37b55", bottom:"#68402a", foam:true, iced:cat==="Iced Coffee"};
}

function reset3DProductPosition(){
  productRotateY = -18;
  productTiltX = 4;
  apply3DProductTransform();
}

function apply3DProductTransform(){
  const drink = $("#drinkModel");
  const pastry = $("#pastryModel");
  const target = (pastry.style.display === "none") ? drink : pastry;
  target.style.transform = `rotateX(${productTiltX}deg) rotateY(${productRotateY}deg)`;
}

function update3DProduct(p){
  const isPastry = p.cat === "Pastries";
  const palette = productPalette(p.name, p.cat);

  $("#drinkModel").style.display = isPastry ? "none" : "block";
  $("#pastryModel").style.display = isPastry ? "block" : "none";

  if (isPastry){
    $("#pastryPiece").style.backgroundImage = `url("${p.image}")`;
    $("#pastryName").textContent = shortTag(p.name);
  } else {
    const drinkModel = $("#drinkModel");
    drinkModel.classList.toggle("iced", palette.iced);
    drinkModel.classList.toggle("hot", !palette.iced);

    $("#frontArt").style.backgroundImage = `url("${p.image}")`;
    $("#productTag").textContent = shortTag(p.name);

    const fillFront = $("#drinkFill");
    const fillBack = $("#drinkFillBack");
    fillFront.style.setProperty("--drinkTop", palette.top);
    fillFront.style.setProperty("--drinkBottom", palette.bottom);
    fillBack.style.setProperty("--drinkTop", palette.top);
    fillBack.style.setProperty("--drinkBottom", palette.bottom);

    $("#drinkFoam").style.display = palette.foam ? "block" : "none";
    $("#drinkFoamBack").style.display = palette.foam ? "block" : "none";
  }

  reset3DProductPosition();
}

$("#addToCart").onclick=()=>{
  if(!selectedProduct)return;
  let add=0;
  if(customization.size==="L") add+=25;
  if(customization.milk!=="Regular") add+=20;
  if(customization.extra==="+ Shot") add+=35;
  if(customization.extra==="+ Syrup") add+=20;
  const item={...selectedProduct,unitPrice:selectedProduct.price+add,custom:{...customization},qty:1};
  const sig=item.id+JSON.stringify(item.custom);
  const existing=state.cart.find(x=>x.sig===sig);
  if(existing) existing.qty++;
  else state.cart.push({...item,sig});
  state.analytics.adds++;save();renderCart();updateAnalytics();closeProduct();toast("Added to cart");
};

function renderCart(){
  $("#cartCount").textContent=state.cart.reduce((a,x)=>a+x.qty,0);
  $("#cartItems").innerHTML=state.cart.length?state.cart.map((x,i)=>`
    <div class="cart-item">
      <div><b>${x.name}</b><br><small>${x.custom.size} • ${x.custom.temp} • ${x.custom.milk} • ${x.custom.extra}</small><br><small>${peso(x.unitPrice)} × ${x.qty}</small></div>
      <div><button class="icon-btn" onclick="changeQty(${i},-1)">−</button> <button class="icon-btn" onclick="changeQty(${i},1)">+</button></div>
    </div>`).join(""):'<p style="color:var(--muted)">Your cart is empty.</p>';
  const total=state.cart.reduce((a,x)=>a+x.unitPrice*x.qty,0);
  $("#cartTotal").textContent=peso(total);
}
window.changeQty=(i,d)=>{state.cart[i].qty+=d;if(state.cart[i].qty<=0)state.cart.splice(i,1);save();renderCart();};

function updateAnalytics(){
  $("#views").textContent=state.analytics.views;
  $("#adds").textContent=state.analytics.adds;
  $("#orders").textContent=state.analytics.orders;
  $("#queueCount").textContent=state.orders.filter(o=>o.status!=="Completed").length;
  $("#adminViews").textContent=state.analytics.views;
  $("#adminAdds").textContent=state.analytics.adds;
  $("#adminOrders").textContent=state.analytics.orders;
  $("#conversion").textContent=(state.analytics.views?Math.round(state.analytics.orders/state.analytics.views*100):0)+"%";
}

}
